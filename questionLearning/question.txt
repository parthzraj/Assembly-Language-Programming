moving display

MVI A,80H  ; puts value in A = 10000000 (08h)
OUT 03H  ; sends content of A into I/o port 03h

MVI A,01H 

START: OUT 01H
       CALL DELAY
       RLC
       JMP START

DELAY: MVI B,FFH

LOOP1: MVI C,FFH

LOOP2: DCR C
       JNZ LOOP2

       DCR B
       JNZ LOOP1

       RET


binary to bcd 

LXI H, 2000H
MOV A, M

MVI B, 0AH ; decimal 10, Because to separate a 
             decimal number into tens and units, we repeatedly subtract 10.

MVI C, 00H

DIVIDE: CMP B
        JC STORE
        SUB B
        INR C
        JMP DIVIDE

STORE: MOV D, A ; A contains ones digit
       MOV A, C ; C contains tens digit
       RLC
       RLC
       RLC
       RLC
       ORA D

       LXI H, 2001H
       MOV M, A
       HLT
